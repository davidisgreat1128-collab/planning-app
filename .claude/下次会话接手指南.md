### 本次重构目标
将 AddTaskPanel.vue 从"混杂架构"重构为严格遵循**四层架构**（Component → Composable → Store → Repository）

### 核心问题
- ❌ Component层混杂业务逻辑（RRULE构建、时间计算）
- ❌ Component直接操作localStorage（分类管理）
- ❌ 复杂业务规则混在UI代码中
- ❌ 无法复用、难以测试

### 重构策略
分6个阶段逐步提取业务逻辑到正确的架构层：
- 阶段1/2: P0级（localStorage + RRULE）
- 阶段3/4/6: P1级（聚合逻辑 + 时间计算 + 日期格式化）
- 阶段5: P2级（时间选择器）

**问题分析**:
`handlePanelSubmit()` 函数（第1209-1269行）包含大量状态同步逻辑：

```javascript
// ❌ 当前实现：30行状态同步逻辑混在Component
async function handlePanelSubmit() {
  // 判断是否有时间段设置
  const hasTimeRange = timeToggle.value && timeStart.value && timeEnd.value;
  const hasDayRange = !timeToggle.value && endDayCount.value > 1;

  form.value.isAllDay = !hasTimeRange;
  form.value.hasTimeRange = hasTimeRange;

  if (hasTimeRange) {
    form.value.startTime = timeStart.value;
    form.value.endTime = timeEnd.value;
  }

  if (hasDayRange) {
    form.value.endDate = endDate.value ? formatDate(endDate.value) : form.value.taskDate;
  }

  // 同步分类字段
  if (selectedCategoryId.value) {
    form.value.planId = selectedCategoryId.value;
  }

  // 同步提醒字段
  if (reminderData.value.enabled) {
    form.value.reminderTime = reminderData.value.time;
  }

  // 同步重复规则字段
  if (repeatData.value && repeatData.value.mode !== 'none') {
    form.value.rrule = buildRrule(repeatData.value);
  }

  await submit();
}
```

**重构方案**:
创建 `composables/useTaskSubmit.js`:

```javascript
export function useTaskSubmit(taskFormApi) {
  const { form, submit: coreSubmit } = taskFormApi;

  async function submitWithPanelData(panelState) {
    // 聚合所有面板状态到form
    const hasTimeRange = panelState.timeToggle && panelState.timeStart && panelState.timeEnd;
    form.value.hasTimeRange = hasTimeRange;
    form.value.isAllDay = !hasTimeRange;

    if (hasTimeRange) {
      form.value.startTime = panelState.timeStart;
      form.value.endTime = panelState.timeEnd;
    }

    // ... 其他同步逻辑

    await coreSubmit();
  }

  return { submitWithPanelData };
}
```

```javascript
// ❌ 当前实现：混在Component（70+行）
const startHour = ref(9);
const startMin = ref(0);
const endHour = ref(18);
const endMin = ref(0);

function adjustStartHour(delta) {
  startHour.value = (startHour.value + delta + 24) % 24;
}

function adjustStartMin(delta) {
  let newMin = startMin.value + delta;
  if (newMin < 0) {
    newMin = 59;
    adjustStartHour(-1);  // 溢出进位逻辑
  } else if (newMin > 59) {
    newMin = 0;
    adjustStartHour(1);
  }
  startMin.value = newMin;
}

// 类似的 adjustEndHour、adjustEndMin...
```

**重构方案**:
创建 `composables/useTimeRangePicker.js`:

```javascript
export function useTimeRangePicker(options = {}) {
  const startHour = ref(options.defaultStartHour || 9);
  const startMin = ref(options.defaultStartMin || 0);
  const endHour = ref(options.defaultEndHour || 18);
  const endMin = ref(options.defaultEndMin || 0);

  // 封装溢出进位逻辑
  function adjustStartHour(delta) { ... }
  function adjustStartMin(delta) { ... }
  function adjustEndHour(delta) { ... }
  function adjustEndMin(delta) { ... }

  const timeStart = computed(() => formatTime(startHour.value, startMin.value));
  const timeEnd = computed(() => formatTime(endHour.value, endMin.value));

  return {
    startHour, startMin, endHour, endMin,
    timeStart, timeEnd,
    adjustStartHour, adjustStartMin, adjustEndHour, adjustEndMin
  };
}
```
