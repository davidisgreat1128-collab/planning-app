**用户场景**：
```
用户长按日历中的重复任务（如"每日晨跑"）
```

## 2. 后端API接口说明

### 2.1 操作4：删除当天

**接口**: `POST /api/v1/tasks/:id/delete-single-day`

**请求参数**:
```json
{
  "date": "2026-03-17"  // 要删除的日期（YYYY-MM-DD）
}
```

**响应数据**:
```json
{
  "success": true,
  "data": {
    "taskId": 123,
    "exdate": ["2026-03-15", "2026-03-17"]  // 更新后的EXDATE数组
  }
}
```

**业务逻辑**（后端已实现）:
1. 验证任务存在且是重复任务
2. 验证日期在RRULE范围内
3. 将日期添加到`task.exdate`数组（不重复添加）
4. 保存Task表
5. 返回更新后的exdate数组

---

### 2.2 操作5：删除全部

**接口**: `DELETE /api/v1/tasks/:id`

**响应数据**:
```json
{
  "success": true,
  "message": "任务已删除"
}
```

**业务逻辑**（后端已实现）:
1. 验证任务存在且是重复任务
2. 软删除任务（设置`deletedAt`字段）
3. 级联删除所有`task_overrides`记录
4. 返回成功消息

---

### 2.3 操作6：删除未来

**接口**: `POST /api/v1/tasks/:id/delete-future`

**请求参数**:
```json
{
  "fromDate": "2026-03-17"  // 从哪天开始删除（YYYY-MM-DD）
}
```

**响应数据**:
```json
{
  "success": true,
  "data": {
    "taskId": 123,
    "rrule": "DTSTART:20260301T000000Z\nRRULE:FREQ=DAILY;UNTIL=20260316T235959Z",
    "rruleUntil": "2026-03-16"
  }
}
```

**业务逻辑**（后端已实现）:
1. 验证任务存在且是重复任务
2. 计算`endDate = fromDate - 1天`（如17号开始删除 → UNTIL=16号）
3. 修改RRULE字符串，添加或更新`UNTIL`参数
4. 更新`task.rrule`和`task.rruleUntil`字段
5. 保存Task表
6. 返回更新后的RRULE

## 3. 前端四层架构实现

### 3.1 Repository层（数据访问层）

**文件**: `frontend/Planning-app/repositories/TaskRepository.js`

**新增3个方法**:

```javascript
/**
 * 删除重复任务的单日实例（添加到EXDATE）⭐ RRULE架构升级（2026-03-17）
 *
 * @param {string} id - 任务 ID
 * @param {string} date - 目标日期（格式：YYYY-MM-DD）
 * @returns {Promise<object>} 更新后的任务对象（包含exdate数组）
 *
 * 实现逻辑：
 * 1. 调用后端API：POST /api/v1/tasks/:id/delete-single-day
 * 2. 后端会：将日期添加到tasks.exdate数组
 * 3. 更新本地缓存（Task表的exdate字段）
 * 4. 通知订阅者触发重新查询
 */
async deleteTaskSingleDay(id, date) {
  const task = this.memoryCache.get(id)
  if (!task) {
    throw new Error(`任务不存在：${id}`)
  }

  console.log('[TaskRepository] 删除单日实例（EXDATE）:', task.title || task.id, '日期:', date)

  try {
    // 调用后端API（POST /tasks/:id/delete-single-day）
    const result = await request({
      url: `/tasks/${id}/delete-single-day`,
      method: 'POST',
      data: { date }
    })

    // 更新本地缓存（更新exdate字段）
    const updatedTask = { ...task, exdate: result.exdate }
    this.memoryCache.set(id, updatedTask)

    // 持久化
    this._saveToLocalStorage()

    // ⭐ 通知订阅者
    this._notify('deleteTaskSingleDay', { taskId: id, date, exdate: result.exdate })

    return result
  } catch (error) {
    console.error('[TaskRepository] 删除单日实例失败:', error)
    throw error
  }
}

/**
 * 删除重复任务的全部实例（软删除任务）⭐ RRULE架构升级（2026-03-17）
 *
 * @param {string} id - 任务 ID
 * @returns {Promise<void>}
 *
 * 实现逻辑：
 * 1. 调用后端API：DELETE /api/v1/tasks/:id
 * 2. 后端会：软删除Task表记录 + 级联删除task_overrides记录
 * 3. 从本地缓存中移除任务
 * 4. 通知订阅者触发UI刷新
 */
async deleteTaskAll(id) {
  const task = this.memoryCache.get(id)
  if (!task) {
    throw new Error(`任务不存在：${id}`)
  }

  console.log('[TaskRepository] 删除全部实例（软删除）:', task.title || task.id)

  try {
    // 调用后端API（DELETE /tasks/:id）
    await request({
      url: `/tasks/${id}`,
      method: 'DELETE'
    })

    // 从内存缓存中移除
    this.memoryCache.delete(id)
    console.log('[TaskRepository] 已从内存缓存中移除:', id)

    // 持久化
    this._saveToLocalStorage()

    // ⭐ 通知订阅者
    this._notify('deleteTaskAll', { taskId: id })
  } catch (error) {
    console.error('[TaskRepository] 删除全部实例失败:', error)
    throw error
  }
}

/**
 * 删除重复任务的未来实例（修改UNTIL）⭐ RRULE架构升级（2026-03-17）
 *
 * @param {string} id - 任务 ID
 * @param {string} fromDate - 从哪天开始删除（格式：YYYY-MM-DD）
 * @returns {Promise<object>} 更新后的任务对象（包含新的rrule和rruleUntil）
 *
 * 实现逻辑：
 * 1. 调用后端API：POST /api/v1/tasks/:id/delete-future
 * 2. 后端会：修改RRULE的UNTIL参数为fromDate前一天
 * 3. 更新本地缓存（Task表的rrule和rruleUntil字段）
 * 4. 通知订阅者触发重新查询
 */
async deleteTaskFuture(id, fromDate) {
  const task = this.memoryCache.get(id)
  if (!task) {
    throw new Error(`任务不存在：${id}`)
  }

  console.log('[TaskRepository] 删除未来实例（修改UNTIL）:', task.title || task.id, '开始日期:', fromDate)

  try {
    // 调用后端API（POST /tasks/:id/delete-future）
    const result = await request({
      url: `/tasks/${id}/delete-future`,
      method: 'POST',
      data: { fromDate }
    })

    // 更新本地缓存（更新rrule和rruleUntil字段）
    const updatedTask = {
      ...task,
      rrule: result.rrule,
      rruleUntil: result.rruleUntil
    }
    this.memoryCache.set(id, updatedTask)

    // 持久化
    this._saveToLocalStorage()

    // ⭐ 通知订阅者
    this._notify('deleteTaskFuture', { taskId: id, fromDate, rrule: result.rrule })

    return result
  } catch (error) {
    console.error('[TaskRepository] 删除未来实例失败:', error)
    throw error
  }
}
```

```javascript
async function confirmDelete() {

  const option = deleteOption.value;
```

## 5. 调用流程

### 5.1 完整调用链

```
用户长按任务
  ↓
Component: onLongPressTask(task)
  ↓
Composable: openDeleteDialog(task)
  ↓
用户选择选项 + 点击"确定"
  ↓
Composable: confirmDelete()
  ↓
根据选项调用不同的Store方法：
  - 选项1 → taskStore.deleteTaskSingleDay(id, date)
  - 选项2 → taskStore.deleteTaskAll(id)
  - 选项3 → taskStore.deleteTaskFuture(id, fromDate)
  ↓
Store: 调用Repository方法
  ↓
Repository: 调用后端API
  ↓
后端API: 处理删除逻辑 + 返回结果
  ↓
Repository: 更新本地缓存 + 通知订阅者
  ↓
Store: 刷新任务列表（fetchTasksByDate）
  ↓
Component: 自动重新渲染（Vue响应式）
```

### 5.2 数据流向

**选项1：仅删除当天**
```
POST /tasks/123/delete-single-day { date: '2026-03-17' }
  ↓
后端: task.exdate = [..., '2026-03-17']
  ↓
Repository: 更新memoryCache的exdate字段
  ↓
Store: 重新获取任务列表
  ↓
Component: 3月17日的任务不显示，其他日期正常显示
```

**选项2：完整清空**
```

  ↓

  ↓

  ↓
Store: 重新获取任务列表
  ↓
Component: 所有日期的该任务都不显示
```

**选项3：删除当天及未来**
```
POST /tasks/123/delete-future { fromDate: '2026-03-17' }
  ↓
后端: task.rrule = '...UNTIL=20260316T235959Z'
  ↓
Repository: 更新memoryCache的rrule和rruleUntil字段
  ↓
Store: 重新获取任务列表
  ↓
Component: 3月17日及之后的任务不显示，3月16日之前正常显示
```

```javascript
if (exdate.length > 50) {

    showCancel: false
  })
  return
}
```

---

### 问题2：删除未来后，UNTIL可能早于DTSTART

**现象**：任务DTSTART=3月10日，用户在3月5日选择"删除未来"

**解决方案**（后端已处理）：
```javascript
// 后端验证逻辑（RecurringTaskService.js）
const dtstart = extractDTSTART(task.rrule)
if (new Date(fromDate) <= new Date(dtstart)) {
  throw new ValidationError('删除起始日期不能早于任务开始日期，请使用"完整清空"')
}
```

---

**现象**：删除后，任务仍显示1-2秒

**解决方案**：
```javascript
// Store层强制立即刷新
async function deleteTaskSingleDay(id, date) {
  await TaskRepository.deleteTaskSingleDay(id, date)

  // ⭐ 强制立即同步（避免防抖延迟）
  await TaskRepository.sync()

  // ⭐ 立即刷新列表
  if (selectedDate.value) {
    await fetchTasksByDate(selectedDate.value)
  }
}
```
