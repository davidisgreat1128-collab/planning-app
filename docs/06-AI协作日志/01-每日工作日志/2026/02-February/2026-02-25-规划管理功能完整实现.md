const diffDays = Math.floor((startDate - today) / (1000 * 60 * 60 * 24));

if (diffDays === 0) return '今天';
if (diffDays === 1) return '明天';
if (diffDays === -1) return '昨天';
if (diffDays > 1) return `${diffDays}天后`;
if (diffDays < -1) return `${Math.abs(diffDays)}天前`;

### 左滑手势识别

```javascript
function onTouchMove(event, categoryId) {
  const touchX = event.touches[0].pageX;
  const touchY = event.touches[0].pageY;
  const deltaX = touchX - touchStartX.value;
  const deltaY = touchY - touchStartY.value;

  // 判断是否是横向滑动
  if (Math.abs(deltaX) > Math.abs(deltaY) && Math.abs(deltaX) > 10) {
    event.preventDefault?.();
  }
}

function onTouchEnd(event, categoryId) {
  const touchX = event.changedTouches[0].pageX;
  const deltaX = touchX - touchStartX.value;

  if (deltaX < -50) {
    swipeOpenId.value = categoryId; // 左滑打开
  } else if (deltaX > 50) {
    swipeOpenId.value = null; // 右滑关闭
  }
}
```
