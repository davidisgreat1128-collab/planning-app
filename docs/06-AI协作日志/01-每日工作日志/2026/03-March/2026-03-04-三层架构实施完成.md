**update(id, changes) - 更新分类（乐观锁）**
```javascript
async update(id, changes) {
  const category = this.memoryCache.get(id)
  if (!category) {
    throw new Error(`分类不存在: ${id}`)
  }

  // 乐观锁：版本号递增
  const updated = {
    ...category,
    ...changes,
    version: category.version + 1,  // 版本号+1
    updatedAt: Date.now()
  }

  // 立即更新内存
  this.memoryCache.set(id, updated)

  // 离线队列记录
  this._addToQueue({
    type: 'update',
    entityId: id,
    data: updated
  })

  this._saveToLocalStorage()
  this._debouncedSync()

  return updated
}
```

**_uploadQueue() - 离线队列上传（指数退避重试）**
```javascript
async _uploadQueue() {
  const pendingOps = this.operationQueue.filter(op => !op.uploaded)

  for (const op of pendingOps) {
    try {
      let response

      // 根据操作类型调用不同 API
      switch (op.type) {
        case 'create':
          response = await createCategory(op.data)
          // 替换临时ID为服务器ID
          this.memoryCache.delete(op.entityId)
          this.memoryCache.set(response.id, response)
          break

        case 'update':
          response = await updateCategory(op.entityId, op.data)
          this.memoryCache.set(op.entityId, response)
          break

        case 'delete':
          await deleteCategory(op.entityId)
          this.memoryCache.delete(op.entityId)
          break
      }

      // 成功：标记已上传
      op.uploaded = true

    } catch (err) {
      console.warn(`[CategoryRepository] 操作上传失败 (第${op.retryCount + 1}次):`, err)

      // 指数退避重试
      op.retryCount = (op.retryCount || 0) + 1

      if (op.retryCount >= 5) {
        // 重试5次后放弃
        console.error('[CategoryRepository] 操作彻底失败，已放弃:', op)
        op.failed = true
      } else {
        // 指数退避：1s → 2s → 4s → 8s → 16s
        const delay = Math.pow(2, op.retryCount) * 1000
        console.log(`[CategoryRepository] ${delay}ms 后重试`)
        await new Promise(resolve => setTimeout(resolve, delay))
      }
    }
  }

  // 清理已上传和已失败的操作
  this.operationQueue = this.operationQueue.filter(
    op => !op.uploaded && !op.failed
  )

  this._saveToLocalStorage()
}
```

**核心实现**：
```javascript
import { defineStore } from 'pinia'
import { computed } from 'vue'
import CategoryRepository from '@/repositories/CategoryRepository'

export const useCategoryStore = defineStore('category', () => {
  // 计算属性 - 自动从 Repository 获取数据
  const categories = computed(() => CategoryRepository.getAll())

  const activeCategoriesCount = computed(() => categories.value.length)

  // Actions - 所有操作委托给 Repository
  async function hydrate() {
    await CategoryRepository.hydrate()
  }

  async function createCategory(data) {
    return await CategoryRepository.create(data)
  }

  async function updateCategory(id, changes) {
    return await CategoryRepository.update(id, changes)
  }

  async function deleteCategory(id) {
    return await CategoryRepository.delete(id)
  }

  async function reorderCategories(newOrder) {
    // 批量更新 sortOrder
    for (const item of newOrder) {
      await CategoryRepository.update(item.id, { sortOrder: item.sortOrder })
    }
  }

  function getCategoryById(id) {
    return CategoryRepository.getById(id)
  }

  async function sync() {
    await CategoryRepository.sync()
  }

  return {
    // State (computed)
    categories,
    activeCategoriesCount,

    // Actions
    hydrate,
    createCategory,
    updateCategory,
    deleteCategory,
    reorderCategories,
    getCategoryById,
    sync
  }
})
```

- ✅ **响应式自动更新**：categories 是 computed 属性，Repository.memoryCache 变化时自动触发视图更新
- ✅ **无冗余状态**：Store 不存储数据副本，Repository 是唯一数据源
- ✅ **薄 Action 层**：所有 action 仅做委托，不包含业务逻辑
- ✅ **易于测试**：可 mock CategoryRepository 进行单元测试

## 💡 技术决策记录

### 决策1：为什么选择 Map 而非 Array 作为 memoryCache？

**问题**：内存缓存应该用什么数据结构？

**选项**：
- A. Array（数组，需要 find() 查找）
- B. Map（映射，键值对）
- C. Object（普通对象）

**决定**：选择 B - Map

**理由**：
1. **性能**：Map 的 get/set/has/delete 都是 O(1)，Array 的 find() 是 O(n)
2. **API 友好**：Map 有 size 属性、values() 迭代器
3. **类型安全**：Map 的 key 可以是任意类型（虽然我们用的是 string）
4. **语义清晰**：id → entity 的映射关系

**代码示例**：
```javascript
// Map 方式（O(1) 查找）
this.memoryCache.set(category.id, category)
const category = this.memoryCache.get(id)  // 快速

// Array 方式（O(n) 查找）- 未采用
this.memoryCache.push(category)
const category = this.memoryCache.find(c => c.id === id)  // 慢
```

### 决策2：为什么使用 500ms debounce 而非立即同步？

**问题**：用户连续修改数据时，何时同步到 localStorage 和服务器？

**选项**：
- A. 立即同步（每次修改都写）
- B. 500ms debounce（500ms 内无新修改才写）
- C. 1000ms debounce

**决定**：选择 B - 500ms debounce

**理由**：
1. **性能优化**：用户拖拽排序时可能连续触发10次 update，debounce 避免10次 localStorage 写入
2. **用户体验**：500ms 足够快，用户感知不到延迟
3. **节省资源**：减少服务器请求次数
4. **数据安全**：500ms 不算长，应用崩溃风险低

**代码示例**：
```javascript
_debouncedSync() {
  clearTimeout(this.syncTimer)
  this.syncTimer = setTimeout(() => {
    this.sync()  // 500ms 后执行
  }, 500)
}
```

### 决策3：为什么采用"本地优先"策略处理版本冲突？

**问题**：乐观锁检测到版本冲突时，应该保留本地修改还是服务器数据？

**选项**：
- A. 服务器优先（覆盖本地修改）
- B. 本地优先（保留用户修改，告警提示）
- C. 弹窗让用户选择

**决定**：选择 B - 本地优先

**理由**：
1. **用户体验**：用户刚修改的数据突然消失，体验极差
2. **单用户场景**：Planning App 是个人应用，多设备编辑概率低
3. **容错性**：即使冲突，用户修改优先，符合心理预期
4. **后续优化空间**：未来可改为选项C（弹窗选择）

**代码示例**：
```javascript
// 冲突检测（简化版，实际代码更复杂）
if (serverVersion > localVersion) {
  console.warn('[Repository] 版本冲突，保留本地修改（本地优先策略）')
  // 保留本地 memoryCache 的数据，不覆盖
} else {
  // 服务器数据更新，合并
  this.memoryCache.set(id, serverData)
}
```

### 决策4：为什么指数退避重试最多5次？

**问题**：离线操作上传失败时，应该重试多少次？

**选项**：
- A. 不重试（失败就放弃）
- B. 无限重试（直到成功）
- C. 重试5次（1s→2s→4s→8s→16s，总计约31秒）

**决定**：选择 C - 重试5次

**理由**：
1. **平衡可靠性与性能**：5次重试覆盖大部分临时网络故障
2. **总时长可接受**：31秒内如果还失败，很可能是服务器宕机或网络完全断开
3. **避免资源浪费**：无限重试会导致内存泄漏、电池耗尽
4. **用户感知**：31秒后仍失败，应该提示用户"服务器无法连接，数据已保存到本地"

**代码示例**：
```javascript
if (op.retryCount >= 5) {
  console.error('[Repository] 操作彻底失败，已放弃:', op)
  op.failed = true
  // 未来可以在 UI 显示"有X条数据未同步"的提示
} else {
  const delay = Math.pow(2, op.retryCount) * 1000  // 指数退避
  await new Promise(resolve => setTimeout(resolve, delay))
}
```

## 🐛 已知问题和注意事项

### 问题1：TaskRepository.getByDate() 时区问题（潜在风险）

**描述**：
```javascript
getByDate(date) {
  return this.getAll().filter(task => {
    if (!task.taskDate) return false
    const taskDate = new Date(task.taskDate).toISOString().split('T')[0]  // ⚠️ UTC时间
    return taskDate === date
  })
}
```

**风险**：
- 如果 `task.taskDate` 是本地时间字符串（如 "2026-03-04"），`new Date()` 会按 UTC 解析
- 不同时区用户可能看到日期偏移1天

**临时方案**：
- 当前数据都是本地生成，格式统一，暂无问题
- 服务器返回 ISO 8601 格式时可能触发

**未来修复**：
- Phase 2 增加 DTO 层，统一处理时区转换

### 问题2：localStorage 5MB 限制未处理

**描述**：
- CategoryRepository 和 TaskRepository 都直接写 localStorage
- 未检测容量，满了会抛出 `QuotaExceededError`

**影响范围**：
- 用户创建大量分类（>1000条）或任务（>5000条）时可能触发

**临时方案**：
- 个人应用，数据量不大，暂时安全

**未来修复**：
- Phase 2 增加 RepositoryFactory，自动切换到 IndexedDB（>50MB）

### 问题3：乐观锁冲突解决策略较简单

**描述**：
- 当前策略：本地优先，直接覆盖服务器数据
- 未记录冲突日志，用户无感知

**风险**：
- 多设备同时编辑同一条数据，可能丢失其中一个设备的修改

**临时方案**：
- Planning App 是个人应用，多设备同时编辑概率极低

**未来修复**：
- Phase 2 增加 ErrorRecovery 层，记录冲突并提供手动合并界面
