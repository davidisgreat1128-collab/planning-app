#### 方法1：`updateTaskSingleDay(id, data, date)`

```javascript
/**
 * 修改重复任务的单日实例（创建task_overrides记录）⭐ RRULE架构升级（2026-03-16）
 *
 * 业务流程：
 * 1. 验证任务存在于缓存
 * 2. 调用后端API：POST /tasks/:id/modify-single-day
 * 3. 后端创建task_overrides记录（NULL字段表示"不覆盖"）
 * 4. 不更新本地Task缓存（因为Task表未改变）
 * 5. 发布事件通知订阅者（触发重新查询getTasksByDate）
 *
 * @param {string} id - 任务ID
 * @param {object} data - 要覆盖的字段（如 { isUrgent: true, isImportant: false }）
 * @param {string} date - 目标日期（格式：YYYY-MM-DD）
 * @returns {Promise<object>} 创建的task_override记录
 */
async updateTaskSingleDay(id, data, date) {
  const task = this.memoryCache.get(id)
  if (!task) {
    throw new Error(`任务不存在：${id}`)
  }

  console.log('[TaskRepository] 修改单日实例（task_overrides）:', task.title || task.id, '日期:', date, '更新:', data)

  try {
    // 调用后端API（POST /tasks/:id/modify-single-day）
    const override = await request({
      url: `/tasks/${id}/modify-single-day`,
      method: 'POST',
      data: {
        date,      // 目标日期
        updates: data  // 要覆盖的字段
      }
    })

    // ⭐ 不更新本地缓存（Task表未改变）
    // ⭐ 但仍然通知订阅者（触发重新查询getTasksByDate）
    this._notify('updateSingleDay', { taskId: id, date, override })

    return override
  } catch (error) {
    console.error('[TaskRepository] 修改单日实例失败:', error)
    throw error
  }
}
```

**关键设计点**：
- ✅ 不更新 `memoryCache`（因为 Task 表未改变，只是创建了 override 记录）
- ✅ 仍然发布 `_notify` 事件（触发 Store 重新查询当天任务）
- ✅ 后端返回创建的 task_override 记录

#### 方法2：`updateTaskFuture(id, data, splitDate)`

```javascript
/**
 * 修改重复任务未来实例（拆分规则）⭐ RRULE架构升级（2026-03-16）
 *
 * 业务流程：
 * 1. 验证任务存在于缓存
 * 2. 调用后端API：POST /tasks/:id/modify-future
 * 3. 后端拆分任务规则：
 *    - 旧任务：添加UNTIL（截止到splitDate前一天）
 *    - 新任务：从splitDate开始，应用新的字段值，parent_task_id指向旧任务
 * 4. 更新本地缓存（旧任务UNTIL已修改）
 * 5. 添加新任务到缓存
 * 6. 持久化到localStorage
 * 7. 发布事件通知订阅者
 *
 * @param {string} id - 任务ID
 * @param {object} data - 要更新的字段（如 { isUrgent: true, isImportant: false }）
 * @param {string} splitDate - 拆分日期（格式：YYYY-MM-DD）
 * @returns {Promise<object>} { oldTask, newTask }
 */
async updateTaskFuture(id, data, splitDate) {
  const task = this.memoryCache.get(id)
  if (!task) {
    throw new Error(`任务不存在：${id}`)
  }

  console.log('[TaskRepository] 修改未来实例（拆分规则）:', task.title || task.id, '拆分日期:', splitDate, '更新:', data)

  try {
    // 调用后端API（POST /tasks/:id/modify-future）
    const result = await request({
      url: `/tasks/${id}/modify-future`,
      method: 'POST',
      data: {
        splitDate,  // 拆分日期
        updates: data  // 要更新的字段
      }
    })

    // result = { oldTask, newTask }
    const { oldTask, newTask } = result

    // 更新旧任务缓存（UNTIL已修改）
    this.memoryCache.set(oldTask.id, oldTask)

    // 添加新任务到缓存
    this.memoryCache.set(newTask.id, newTask)

    // 持久化
    this._saveToLocalStorage()

    // ⭐ 发布事件：通知订阅者任务已更新
    this._notify('updateTaskFuture', { oldTask, newTask })

    return result
  } catch (error) {
    console.error('[TaskRepository] 修改未来实例失败:', error)
    throw error
  }
}
```

**关键设计点**：
- ✅ 更新 `memoryCache` 中的旧任务（UNTIL 已修改）
- ✅ 添加新任务到 `memoryCache`
- ✅ 持久化到 `localStorage`
- ✅ 发布事件触发 UI 更新

**实现过程中的技术挑战**：

**背景**：前端和后端方法名不一致

**决策**：
- ✅ 统一使用 `updateTaskSingleDay`（不使用 `updateOnce` 或 `modifySingleDay`）
- ✅ 统一使用 `updateTaskFuture`（不使用 `updateFuture` 或 `modifyFuture`）

**理由**：

2. 符合6操作命名规范（文档中的操作名）
3. 语义更清晰（"SingleDay"比"Once"更明确）

### 决策2：Store层必须添加刷新逻辑

**背景**：修改任务后UI不会自动更新

**决策**：
- ✅ 在 `updateTaskFuture()` 和 `updateTaskSingleDay()` 方法中添加 `await fetchTasksByDate(selectedDate.value)`

**理由**：
1. task_overrides 记录不会触发 Repository 的 memoryCache 更新
2. 任务拆分会创建新任务，需要重新查询
3. 用户操作后应立即看到UI变化（UX要求）

**实现方式**：
```javascript
async function updateTaskSingleDay(id, data, date) {
  const override = await TaskRepository.updateTaskSingleDay(id, data, date)

  // ⭐ 关键：刷新当前日期的任务列表
  if (selectedDate.value) {
    await fetchTasksByDate(selectedDate.value)
  }

  return override
}
```

### 决策3：Repository层事件通知策略

**背景**：`updateTaskSingleDay` 不更新 memoryCache，但仍需要通知订阅者

**决策**：
- ✅ 调用 `this._notify('updateSingleDay', ...)`（即使没有更新缓存）

**理由**：
1. Store 层监听此事件，会主动调用 `fetchTasksByDate()` 刷新数据
2. 保持事件驱动架构的一致性
3. 便于未来扩展（如添加其他订阅者）
