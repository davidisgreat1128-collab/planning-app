  console.log('[useDragDrop] onTaskTouchEnd 被调用, dragging:', dragState.value.dragging);

  if (!dragState.value.dragging) return;

  e.preventDefault?.();
  e.stopPropagation?.();

  console.log('[useDragDrop] onTaskTouchEnd - 状态:', {
    task: task?.title,
    fromQuadrant,
    overDelete,
    position: { x, y }
  });

  // 重置拖拽状态
  dragState.value.dragging = false;

  // 如果在删除区域上方,返回标识让调用方显示删除对话框
  if (overDelete) {
    console.log('[useDragDrop] onTaskTouchEnd - 检测到在删除区域,触发删除');
    onDragEnd(task, fromQuadrant, null, { shouldDelete: true });
    return;
  }

  // 检测拖拽到哪个象限
  const target = detectQuadrantAtPosition(x, y);
  console.log('[useDragDrop] onTaskTouchEnd - 检测到目标象限:', target);

  if (target && target !== fromQuadrant) {
    // 通知调用方处理象限变更
    console.log('[useDragDrop] onTaskTouchEnd - 触发象限变更:', fromQuadrant, '→', target);
    onDragEnd(task, fromQuadrant, target, { shouldDelete: false });
  } else {
    // 未移动或回到原象限,取消拖拽
    console.log('[useDragDrop] onTaskTouchEnd - 无效移动,取消拖拽');

async function changeTaskQuadrant(task, from, to, selectedDate) {
  console.log('[useTaskQuadrant] changeTaskQuadrant 被调用:', {
    task: task?.title,
    from,
    to,
    selectedDate
  });

  if (from === to) {
    console.log('[useTaskQuadrant] changeTaskQuadrant - 象限相同,无需更改');
    return;
  }

  // 检查任务是否为重复任务
  const isRecurring = task.isRecurring || task.rrule;
  console.log('[useTaskQuadrant] changeTaskQuadrant - 任务类型:', isRecurring ? '重复任务' : '普通任务');

  if (isRecurring) {
    // 重复任务:显示对话框让用户选择
    console.log('[useTaskQuadrant] changeTaskQuadrant - 打开重复任务对话框');
    openChangeQuadrantDialog(task, from, to);
  } else {
    // 普通任务:直接更改象限
    const { isUrgent, isImportant } = getQuadrantProperties(to);
    console.log('[useTaskQuadrant] changeTaskQuadrant - 更新普通任务象限:', { isUrgent, isImportant });

    try {
      await taskStore.updateTask(task.id, { isUrgent, isImportant });
      await taskStore.fetchTasksByDate(selectedDate);
      console.log('[useTaskQuadrant] changeTaskQuadrant - 更新成功');
      uni.showToast({ title: '已更改', icon: 'success' });
    } catch (err) {
      console.error('[useTaskQuadrant] 更改象限失败:', err);
      uni.showToast({ title: '更改失败', icon: 'none' });
    }
  }
}

  console.log('[CalendarBar] onTouchEnd - 状态:', {
    touchMoved,
    touchDir,
    mode: props.mode
  });

  if (!touchMoved || touchDir === '') {
    console.log('[CalendarBar] onTouchEnd - 未达到方向判断阈值,视为点击');
    return;
  }

  const dx = e.changedTouches[0].clientX - touchStartX;
  const dy = e.changedTouches[0].clientY - touchStartY;

  console.log('[CalendarBar] onTouchEnd - 滑动距离:', { dx, dy });

  if (touchDir === 'h' && Math.abs(dx) > 40) {
    // 水平滑动: 切换周/月
    if (dx < 0) {
      console.log('[CalendarBar] onTouchEnd - 触发左滑 (下一周/月)');
      emit('swipe-left');
    } else {
      console.log('[CalendarBar] onTouchEnd - 触发右滑 (上一周/月)');
      emit('swipe-right');
    }
  } else if (touchDir === 'v') {
    if (props.mode === 'week' && dy > 50) {
      console.log('[CalendarBar] onTouchEnd - 触发展开月视图');
      emit('expand');
    } else if (props.mode === 'month' && dy < -50) {
      console.log('[CalendarBar] onTouchEnd - 触发折叠回周视图');
      emit('collapse');
    }
  }
}

### 常见问题诊断

#### 问题1: 日历条完全不显示

**可能原因**:
1. currentWeekDates 计算属性返回空数组
2. CalendarBar 组件未挂载
3. CSS 样式问题 (高度为0、display:none)

**诊断日志**:
```
[useCalendar] currentWeekDates 计算属性被调用, currentWeekStart: null
[useCalendar] currentWeekStart 为空,返回空数组
```

**解决方案**:
- 检查 useCalendar.init() 是否被调用
- 检查 currentWeekStart.value 是否正确设置

---

#### 问题2: 节日/农历不显示

**可能原因**:
1. loadHolidays() 未被调用
2. API 请求失败
3. holidayMap 未正确填充

**诊断日志**:
```
[useCalendar] _loadHolidayRange 错误: Error: ...
```

**解决方案**:
- 检查后端服务是否启动
- 检查 API 路由是否正确
- 查看 Network 面板中的请求响应

---

#### 问题3: 手势滑动不响应

**可能原因**:
1. touch 事件未绑定
2. 滑动距离未达到阈值
3. 事件被其他元素拦截

**诊断日志**:
```
[CalendarBar] onTouchStart - 起始坐标: { x: 100, y: 200, mode: 'week' }
[CalendarBar] onTouchEnd - 状态: { touchMoved: false, touchDir: '', mode: 'week' }
[CalendarBar] onTouchEnd - 未达到方向判断阈值,视为点击
```

**解决方案**:
- 确保滑动距离 > 40px (水平) 或 > 50px (垂直)
- 检查 CSS 中是否有 `touch-action: none` 样式
- 尝试在触摸屏设备上测试,PC鼠标模拟可能不准确

---

#### 问题4: 任务拖拽不工作

**可能原因**:
1. 长按未触发 (< 500ms)
2. dragState 未正确更新
3. 目标象限检测失败

**诊断日志**:
```
[useDragDrop] onTaskLongPress 被触发, 任务: 完成报告
[useDragDrop] startDrag 被调用, 任务: 完成报告, 象限: q1
[useDragDrop] startDrag - 起始坐标: { x: 150, y: 300 }
[useDragDrop] startDrag - dragState 已更新: { dragging: true, ... }
[useDragDrop] onTaskTouchEnd 被调用, dragging: true
[useDragDrop] onTaskTouchEnd - 检测到目标象限: q2
[useDragDrop] onTaskTouchEnd - 触发象限变更: q1 → q2
```

**解决方案**:
- 确保长按任务 > 500ms
- 检查 dragState.value.dragging 是否为 true
- 验证 detectQuadrantAtPosition() 函数是否正确检测坐标

### 日志添加的最佳实践

**原则1: 在关键路径的入口和出口添加日志**
```javascript
// ✅ 好的做法
async function loadData() {
  console.log('[module] loadData 开始');
  try {
    const data = await fetchAPI();
    console.log('[module] loadData 完成, 数据:', data);
    return data;
  } catch (err) {
    console.error('[module] loadData 失败:', err);
    throw err;
  }
}

// ❌ 不好的做法 (缺少完成日志)
async function loadData() {
  console.log('[module] loadData 开始');
  const data = await fetchAPI();
  return data;
}
```

**原则2: 记录关键状态变更**
```javascript
// ✅ 好的做法
function setState(newState) {
  console.log('[module] setState - 旧状态:', state.value);
  state.value = newState;
  console.log('[module] setState - 新状态:', state.value);
}

// ❌ 不好的做法 (缺少状态信息)
function setState(newState) {
  console.log('[module] setState 被调用');
  state.value = newState;
}
```

**原则3: 使用结构化数据**
```javascript
// ✅ 好的做法
console.log('[module] API响应:', { status, data, timestamp });

// ❌ 不好的做法 (字符串拼接)
console.log('[module] API响应: status=' + status + ', data=' + data);
```

**原则4: 避免日志污染**
```javascript
// ✅ 好的做法 (关键操作才打日志)
function handleClick() {
  console.log('[component] handleClick - 任务:', task.title);
  emit('task-click', task);
}

// ❌ 不好的做法 (高频操作打日志)
function onMouseMove(e) {
  console.log('[component] onMouseMove - x:', e.clientX); // ❌ 会刷屏
}
```
